using System.Runtime.InteropServices;
namespace LSE.InventorFootprintTools;
public static class Ace {
 public static dynamic Open(string path){if(!File.Exists(path))throw new FileNotFoundException("Access database was not found.",path);Type t=Type.GetTypeFromProgID("ADODB.Connection")??throw new InvalidOperationException("ADODB unavailable.");dynamic c=Activator.CreateInstance(t)!;var errors=new List<string>();foreach(string p in new[]{"Microsoft.ACE.OLEDB.16.0","Microsoft.ACE.OLEDB.12.0","Microsoft.Jet.OLEDB.4.0"})try{c.Open($"Provider={p};Data Source={path};Persist Security Info=False;");return c;}catch(Exception ex){errors.Add(p+": "+ex.Message);}Release(c);throw new InvalidOperationException("No 64-bit Access OLEDB provider could open "+path+Environment.NewLine+string.Join(Environment.NewLine,errors));}
 public static List<string> Tables(dynamic c){dynamic r=null;try{r=c.OpenSchema(20);var x=new List<string>();while(!r.EOF){string n=Convert.ToString(r.Fields[2].Value)??"",k=Convert.ToString(r.Fields[3].Value)??"";if(k.Equals("TABLE",StringComparison.OrdinalIgnoreCase)&&!n.StartsWith("MSys",StringComparison.OrdinalIgnoreCase))x.Add(n);r.MoveNext();}return x;}finally{Release(r);}}
 public static void Exec(dynamic c,string sql){dynamic r=null;try{r=c.Execute(sql);}finally{Release(r);}}
 public static string E(string v)=>v.Replace("'","''");
 public static void ValidateTable(string t){if(!System.Text.RegularExpressions.Regex.IsMatch(t,@"^[A-Za-z0-9_][A-Za-z0-9_ &/\.\-]*$"))throw new InvalidOperationException("Invalid table name: "+t);}
 public static void Release(object? o){if(o!=null&&Marshal.IsComObject(o))try{Marshal.FinalReleaseComObject(o);}catch{}}
 public static void Close(dynamic? c){if(c is null)return;try{c.Close();}catch{}Release(c);}
}
public sealed class AccessCatalogService {
 readonly string path; public AccessCatalogService(string p)=>path=p;
 public List<string> GetTables(){dynamic c=Ace.Open(path);try{return ((List<string>)Ace.Tables(c)).Where(n=>!n.StartsWith("_")).OrderBy(n=>n,StringComparer.OrdinalIgnoreCase).ToList();}finally{Ace.Close(c);}}
 public Dictionary<string,string> GetFamilyDescriptions(){var x=new Dictionary<string,string>(StringComparer.OrdinalIgnoreCase);dynamic c=Ace.Open(path),r=null;try{r=c.Execute("SELECT FAMILY,[DESC] FROM [_FAMILY_DESCRIPTION]");while(!r.EOF){string f=Convert.ToString(r.Fields[0].Value)??"";object v=r.Fields[1].Value;if(f.Length>0)x[f.Trim()]=v is null or DBNull?"":Convert.ToString(v)?.Trim()??"";r.MoveNext();}}catch{}finally{Ace.Release(r);Ace.Close(c);}return x;}
 public List<string> ReadUser1StartingWith(string prefix){var x=new List<string>();dynamic c=Ace.Open(path);try{foreach(string t in ((List<string>)Ace.Tables(c)).Where(n=>!n.StartsWith("_"))){dynamic r=null;try{r=c.Execute($"SELECT USER1 FROM [{t}] WHERE USER1 LIKE '{Ace.E(prefix)}%'");while(!r.EOF){object v=r.Fields[0].Value;if(v is not null and not DBNull)x.Add(Convert.ToString(v)??"");r.MoveNext();}}catch{}finally{Ace.Release(r);}}return x;}finally{Ace.Close(c);}}
 public bool Exists(string table,string mfg,string cat){Ace.ValidateTable(table);dynamic c=Ace.Open(path),r=null;try{r=c.Execute($"SELECT TOP 1 CATALOG FROM [{table}] WHERE MANUFACTURER='{Ace.E(mfg)}' AND CATALOG='{Ace.E(cat)}'");return !(bool)r.EOF;}finally{Ace.Release(r);Ace.Close(c);}}
 public void AddRecord(CatalogAddRequest q){Ace.ValidateTable(q.TableName);dynamic c=Ace.Open(path),schema=null;try{schema=c.Execute($"SELECT TOP 1 * FROM [{q.TableName}]");var cols=new HashSet<string>(StringComparer.OrdinalIgnoreCase);for(int i=0;i<schema.Fields.Count;i++)cols.Add(Convert.ToString(schema.Fields[i].Name)??"");Ace.Release(schema);schema=null;if(!cols.Contains("USER1"))throw new InvalidOperationException($"Table {q.TableName} has no USER1 column, so it cannot hold an LSE part number.");
  var values=new Dictionary<string,string>(StringComparer.OrdinalIgnoreCase){{"CATALOG",q.Catalog},{"MANUFACTURER",q.Manufacturer},{"DESCRIPTION",q.Description},{"TYPE",q.Type},{"RATING",q.Rating},{"USER1",q.User1},{"USER2",q.User2},{"USER3",q.User3},{"TEXTVALUE",q.TextValue},{"WEBLINK",q.WebLink},{"WDBLKNAM",q.WdBlkNam}};
  var use=values.Where(k=>cols.Contains(k.Key)&&k.Value.Length>0).ToList();string names=string.Join(",",use.Select(k=>$"[{k.Key}]"));string vals=string.Join(",",use.Select(k=>$"'{Ace.E(k.Value)}'"));Ace.Exec(c,$"INSERT INTO [{q.TableName}] ({names}) VALUES ({vals})");
 }finally{Ace.Release(schema);Ace.Close(c);}}
}
public sealed class FootprintLookupService {
 readonly string path; public FootprintLookupService(string p)=>path=p;
 public bool TableExists(string t){dynamic c=Ace.Open(path);try{return ((List<string>)Ace.Tables(c)).Any(n=>n.Equals(t,StringComparison.OrdinalIgnoreCase));}finally{Ace.Close(c);}}
 public void EnsureTable(string t){Ace.ValidateTable(t);if(TableExists(t))return;dynamic c=Ace.Open(path);try{Ace.Exec(c,$"CREATE TABLE [{t}] ([CATALOG] TEXT(60), [ASSEMBLYCODE] TEXT(24), [BLKNAM] TEXT(255), [DESCRIPTION] TEXT(150), [RECNUM] COUNTER)");}finally{Ace.Close(c);}}
 public string? Find(string t,string catalog){Ace.ValidateTable(t);dynamic c=Ace.Open(path),r=null;try{r=c.Execute($"SELECT TOP 1 BLKNAM FROM [{t}] WHERE CATALOG='{Ace.E(catalog)}' AND (ASSEMBLYCODE IS NULL OR ASSEMBLYCODE='')");if((bool)r.EOF)return null;object v=r.Fields[0].Value;return v is null or DBNull?"":Convert.ToString(v);}finally{Ace.Release(r);Ace.Close(c);}}
 public void Upsert(string t,string catalog,string blkName,string description){Ace.ValidateTable(t);if(blkName.Length>255)throw new InvalidOperationException("Footprint path is longer than 255 characters (BLKNAM limit): "+blkName);if(description.Length>150)description=description[..150];dynamic c=Ace.Open(path);try{if(Find(t,catalog) is not null)Ace.Exec(c,$"UPDATE [{t}] SET [BLKNAM]='{Ace.E(blkName)}',[DESCRIPTION]='{Ace.E(description)}' WHERE CATALOG='{Ace.E(catalog)}' AND (ASSEMBLYCODE IS NULL OR ASSEMBLYCODE='')");else Ace.Exec(c,$"INSERT INTO [{t}] ([CATALOG],[BLKNAM],[DESCRIPTION]) VALUES ('{Ace.E(catalog)}','{Ace.E(blkName)}','{Ace.E(description)}')");}finally{Ace.Close(c);}}
}
